// CLI共通起動のUDP可否、HTTP/2継続、HTTP/3実通信と証明書再読込を検査する。
package main

import (
	"context"
	"crypto/tls"
	"io"
	"net"
	"net/http"
	"testing"
	"time"

	"github.com/quic-go/quic-go/http3"
	"oimobbs/internal/bootstrap"
)

func TestServeHTTP3StartupAndUDPBusy(t *testing.T) {
	for _, mode := range []string{"disabled", "busy", "available"} {
		t.Run(mode, func(t *testing.T) {
			cfg, client, rotate := publicFixture(t)
			cfg.Public.HTTP3 = mode != "disabled"
			if _, err := bootstrap.Initialize(cfg, "HTTP/3 startup"); err != nil {
				t.Fatal(err)
			}
			if mode == "busy" {
				socket, err := net.ListenPacket("udp", cfg.Public.Listen)
				if err != nil {
					t.Fatal(err)
				}
				defer socket.Close()
			}
			ctx, cancel := context.WithCancel(t.Context())
			defer cancel()
			done := make(chan error, 1)
			go func() {
				done <- serveConfiguredWithNetworkClient(ctx, cfg, false, "", "", "", io.Discard, false, nil, false, offlineNetworkClient())
			}()
			deadline := time.Now().Add(10 * time.Second)
			var res *http.Response
			for {
				var err error
				res, err = client.Get(cfg.Public.BaseURL + "/bbsmenu.json")
				if err == nil {
					res.Body.Close()
					if res.StatusCode == 200 {
						break
					}
				}
				if time.Now().After(deadline) {
					t.Fatal("TCP server not ready", err)
				}
				select {
				case err := <-done:
					t.Fatal("startup exited", err)
				case <-time.After(5 * time.Millisecond):
				}
			}
			if res.ProtoMajor != 2 {
				t.Fatalf("HTTP/2 fallback: %s", res.Proto)
			}
			if mode == "available" {
				if res.Header.Get("Alt-Svc") == "" {
					t.Fatal("active HTTP/3 not advertised")
				}
				tlsConfig := client.Transport.(*http.Transport).TLSClientConfig.Clone()
				tlsConfig.MinVersion = tls.VersionTLS13
				for _, serial := range []int64{1, 2} {
					if serial == 2 {
						rotate(serial)
					}
					tr := &http3.Transport{TLSClientConfig: tlsConfig}
					h3client := &http.Client{Transport: tr, Timeout: 3 * time.Second}
					res, err := h3client.Get(cfg.Public.BaseURL + "/bbsmenu.json")
					if err != nil {
						tr.Close()
						t.Fatal(err)
					}
					body, err := io.ReadAll(res.Body)
					res.Body.Close()
					tr.Close()
					if err != nil || len(body) == 0 || res.StatusCode != 200 || res.ProtoMajor != 3 || res.TLS.PeerCertificates[0].SerialNumber.Int64() != serial {
						t.Fatalf("HTTP/3 certificate=%d: %v %v", serial, res, err)
					}
				}
			} else if res.Header.Get("Alt-Svc") != "" {
				t.Fatal("unavailable HTTP/3 advertised")
			}
			cancel()
			select {
			case err := <-done:
				if err != nil {
					t.Fatal(err)
				}
			case <-time.After(5 * time.Second):
				t.Fatal("shutdown did not drain")
			}
			report, err := bootstrap.Check(cfg)
			if err != nil {
				t.Fatal(err)
			}
			if (report.MinimumReader >= 107) != (mode == "available") {
				t.Fatalf("reader changed without active UDP: %d", report.MinimumReader)
			}
			if mode == "available" {
				socket, err := net.ListenPacket("udp", cfg.Public.Listen)
				if err != nil {
					t.Fatal("UDP leaked after shutdown", err)
				}
				socket.Close()
			}
		})
	}
}
