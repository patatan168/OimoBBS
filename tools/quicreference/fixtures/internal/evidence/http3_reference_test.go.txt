// HTTP/3の記録にUDPとTLS1.3が揃うことと、従来TCPの検証境界を検査する。
package evidence

import (
	"crypto/tls"
	"github.com/quic-go/quic-go/http3"
	"io"
	"net"
	"net/http"
	"net/http/httptest"
	"oimobbs/internal/domain"
	"oimobbs/internal/httptransport"
	"strings"
	"testing"
	"time"
)

func TestHTTP3RealConnectionObservation(t *testing.T) {
	testhttp3realconnectionobservation(t, httptransport.Prepare)
}
func testhttp3realconnectionobservation(t *testing.T, prepare func(*http.Server, net.Addr, string, bool, int, int) (*httptransport.HTTP3, string, error)) {
	observer := testObserver(t)
	owner := domain.NewID()
	seen := make(chan HTTPObservation, 2)
	tcp := httptest.NewUnstartedServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		capture, err := observer.Begin(r, owner, "bbs_post", "example.test")
		if err != nil {
			t.Error(err)
			w.WriteHeader(500)
			return
		}
		if _, err = io.Copy(io.Discard, r.Body); err != nil {
			t.Error(err)
			return
		}
		observation, err := capture.Finish()
		if err != nil {
			t.Error(err)
			w.WriteHeader(500)
			return
		}
		seen <- observation
		w.WriteHeader(204)
	}))
	tcp.Listener.Close()
	listener, err := net.Listen("tcp", "0.0.0.0:0")
	if err != nil {
		t.Fatal(err)
	}
	tcp.Listener = listener
	_, port, _ := net.SplitHostPort(listener.Addr().String())
	address := net.JoinHostPort("127.0.0.1", port)
	defer tcp.Close()
	cert := httptest.NewTLSServer(nil)
	defer cert.Close()
	tcp.Config.TLSConfig = cert.TLS.Clone()
	tcp.Config.ConnContext = observer.ConnContext
	h3, reason, err := prepare(tcp.Config, tcp.Listener.Addr(), "https://"+address, true, 16, 4)
	if err != nil || h3 == nil {
		t.Fatalf("%s %v", reason, err)
	}
	defer h3.Close()
	go h3.Serve()
	tr := &http3.Transport{TLSClientConfig: &tls.Config{InsecureSkipVerify: true}}
	defer tr.Close()
	client := &http.Client{Transport: tr, Timeout: 3 * time.Second}
	for range 2 {
		res, err := client.Post("https://"+address, "application/x-www-form-urlencoded", strings.NewReader("body=hello"))
		if err != nil {
			t.Fatal(err)
		}
		res.Body.Close()
		if res.StatusCode != 204 {
			t.Fatal(res.Status)
		}
	}
	first, second := <-seen, <-seen
	if first.Transport != "udp" || first.HTTPVersion != "HTTP/3.0" || first.TLSVersion == nil || *first.TLSVersion != tls.VersionTLS13 || first.ConnectionID != second.ConnectionID || first.RequestID == second.RequestID || first.Peer.IP != "127.0.0.1" || first.Peer.Port == 0 || first.Local.Port == 0 {
		t.Fatalf("UDP observation: %+v %+v", first, second)
	}
	if err := ValidateHTTPObservationValue(first); err != nil {
		t.Fatal(err)
	}
}
