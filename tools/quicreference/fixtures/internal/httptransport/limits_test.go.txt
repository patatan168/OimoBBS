// 遅いHTTP/3ヘッダーを打ち切り、同じ接続の別要求を継続できることを検査する。
package httptransport

import (
	"bytes"
	"context"
	"crypto/tls"
	"errors"
	"fmt"
	"io"
	"net/http"
	"net/http/httptest"
	"runtime"
	"strconv"
	"testing"
	"time"

	"github.com/quic-go/quic-go"
	"github.com/quic-go/quic-go/http3"
)

func TestHTTP3SlowHeaderDoesNotBlockOtherStream(t *testing.T) {
	tcp := httptest.NewUnstartedServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { io.WriteString(w, "ok") }))
	defer tcp.Close()
	cert := httptest.NewTLSServer(nil)
	defer cert.Close()
	tcp.Config.TLSConfig = cert.TLS.Clone()
	tcp.Config.ReadHeaderTimeout = 50 * time.Millisecond
	tcp.Config.ReadTimeout = time.Second
	tcp.Config.WriteTimeout = time.Second
	h, reason, err := Prepare(tcp.Config, tcp.Listener.Addr(), "https://"+tcp.Listener.Addr().String(), true, 16, 4)
	if err != nil || h == nil {
		t.Fatalf("%s %v", reason, err)
	}
	defer h.Close()
	go h.Serve()
	ctx, cancel := context.WithTimeout(t.Context(), 3*time.Second)
	defer cancel()
	conn, err := quic.DialAddr(ctx, tcp.Listener.Addr().String(), &tls.Config{InsecureSkipVerify: true, NextProtos: []string{http3.NextProtoH3}}, nil)
	if err != nil {
		t.Fatal(err)
	}
	defer conn.CloseWithError(0, "")
	stream, err := conn.OpenStreamSync(ctx)
	if err != nil {
		t.Fatal(err)
	}
	// HEADERSの長さ10を宣言し、一部だけ送る。
	if _, err = stream.Write([]byte{1, 10, 0}); err != nil {
		t.Fatal(err)
	}
	stream.SetReadDeadline(time.Now().Add(time.Second))
	_, err = stream.Read(make([]byte, 1))
	var reset *quic.StreamError
	if !errors.As(err, &reset) {
		t.Fatalf("slow headers not reset: %v", err)
	}
	tr := &http3.Transport{}
	defer tr.Close()
	client := tr.NewClientConn(conn)
	req, _ := http.NewRequestWithContext(ctx, "GET", "https://"+tcp.Listener.Addr().String(), nil)
	res, err := client.RoundTrip(req)
	if err != nil {
		t.Fatal(err)
	}
	body, err := io.ReadAll(res.Body)
	res.Body.Close()
	if err != nil || res.StatusCode != 200 || string(body) != "ok" {
		t.Fatalf("other stream: %s %v", body, err)
	}
}

func TestHTTP3ShutdownDrainsResponse(t *testing.T) {
	entered := make(chan struct{})
	release := make(chan struct{})
	body := bytes.Repeat([]byte("drain"), 64<<10)
	tcp := httptest.NewUnstartedServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		close(entered)
		<-release
		w.Header().Set("Content-Length", strconv.Itoa(len(body)))
		w.Write(body)
	}))
	defer tcp.Close()
	cert := httptest.NewTLSServer(nil)
	defer cert.Close()
	tcp.Config.TLSConfig = cert.TLS.Clone()
	tcp.Config.ReadHeaderTimeout = time.Second
	tcp.Config.WriteTimeout = time.Second
	h, reason, err := Prepare(tcp.Config, tcp.Listener.Addr(), "https://"+tcp.Listener.Addr().String(), true, 16, 4)
	if err != nil || h == nil {
		t.Fatalf("%s %v", reason, err)
	}
	defer h.Close()
	go h.Serve()
	tr := &http3.Transport{TLSClientConfig: &tls.Config{InsecureSkipVerify: true}}
	defer tr.Close()
	client := &http.Client{Transport: tr, Timeout: 3 * time.Second}
	response := make(chan error, 1)
	go func() {
		res, err := client.Get("https://" + tcp.Listener.Addr().String())
		if err == nil {
			var got []byte
			got, err = io.ReadAll(res.Body)
			res.Body.Close()
			if err == nil && !bytes.Equal(got, body) {
				err = fmt.Errorf("truncated: %d", len(got))
			}
		}
		response <- err
	}()
	<-entered
	ctx, cancel := context.WithTimeout(t.Context(), 3*time.Second)
	defer cancel()
	stopped := make(chan error, 1)
	go func() { stopped <- h.Shutdown(ctx) }()
	for {
		h.mu.Lock()
		closing := h.closing
		h.mu.Unlock()
		if closing {
			break
		}
		runtime.Gosched()
	}
	close(release)
	if err := <-response; err != nil {
		t.Fatal("response during shutdown", err)
	}
	// 応答をすべて受け取ったpeerが接続を終了してから、停止完了を確認する。
	tr.Close()
	if err := <-stopped; err != nil {
		t.Fatal(err)
	}
}
