// 実HTTP/3診断の終了後、qlogの最終recordまで有効なJSONとして保存されることを検査する。
package main

import (
	"bytes"
	"encoding/json"
	"io"
	"net/http"
	"net/http/httptest"
	"os"
	"path/filepath"
	"testing"
	"time"

	"oimobbs/internal/httptransport"
)

func TestClientTraceDrainsBeforeExit(t *testing.T) {
	traceDir = t.TempDir()
	defer func() { traceDir = "" }()
	tcp := httptest.NewUnstartedServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { w.WriteHeader(http.StatusNotModified) }))
	defer tcp.Close()
	cert := httptest.NewTLSServer(nil)
	tcp.Config.TLSConfig = cert.TLS.Clone()
	cert.Close()
	h3, reason, err := httptransport.Prepare(tcp.Config, tcp.Listener.Addr(), "https://"+tcp.Listener.Addr().String(), true, 16, 4)
	if err != nil || h3 == nil {
		t.Fatalf("%s %v", reason, err)
	}
	defer h3.Close()
	go h3.Serve()
	client, stop, err := newClient("h3")
	if err != nil {
		t.Fatal(err)
	}
	client.Timeout = 3 * time.Second
	defer stop()
	for range 3 {
		res, err := client.Get("https://" + tcp.Listener.Addr().String())
		if err != nil {
			t.Fatal(err)
		}
		_, err = io.Copy(io.Discard, res.Body)
		res.Body.Close()
		if err != nil || res.StatusCode != 304 {
			t.Fatalf("%v %v", res, err)
		}
	}
	stop()
	traceWait.Wait()
	paths, err := filepath.Glob(filepath.Join(traceDir, "*.qlog"))
	if err != nil || len(paths) != 1 {
		t.Fatal(paths, err)
	}
	raw, err := os.ReadFile(paths[0])
	if err != nil {
		t.Fatal(err)
	}
	parsed := 0
	for _, record := range bytes.Split(raw, []byte{0x1e}) {
		if len(bytes.TrimSpace(record)) == 0 {
			continue
		}
		var event struct {
			Name string `json:"name"`
			Data struct {
				Frame struct {
					Type string `json:"frame_type"`
				} `json:"frame"`
			} `json:"data"`
		}
		if err := json.Unmarshal(record, &event); err != nil {
			t.Fatal("truncated qlog", err)
		}
		if event.Name == "http3:frame_parsed" && event.Data.Frame.Type == "headers" {
			parsed++
		}
	}
	if parsed != 3 {
		t.Fatalf("response headers missing from qlog: %d", parsed)
	}
}
